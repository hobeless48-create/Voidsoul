[ITEM STRUCTURE]
interface Item {
  id: string;              // Unique item ID
  name: string;            // Display name
  type: "weapon"|"offhand"|"helmet"|"chestplate"|"leggings"|"boots"|"ring"|"bracelet"|"necklace"|"charm"|"material"|"consumable";
  flatAtk?: number;        // Flat ATK bonus
  flatMaxHp?: number;      // Flat HP bonus
  flatCrit?: number;       // +X% crit chance
  flatDodge?: number;      // +X% dodge
  flatDef?: number;        // Flat DEF bonus
  flatAccuracy?: number;   // +X% accuracy
  weaponType?: "sword"|"bow"|"staff"|"tome"|"dagger"|"lute"|"shield";
  statsBonus?: Partial<Stats>; // STR,DEX,CON,INT,FTH,CHA,LCK bonuses
  description: string;     // Flavor text with special effects
  value: number;           // Buy/sell price
  quantity?: number;       // Stack count
  ownerId?: string;        // P2P owner tracking
  isTwoHanded?: boolean;   // Uses both weapon + offhand slots
  rarity?: "Common"|"Uncommon"|"Rare"|"Epic"|"Legendary";
}
