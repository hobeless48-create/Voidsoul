[ENEMY TEMPLATE STRUCTURE]
interface EnemyTemplate {
  name: string;           // Display name
  maxHp: number;          // Base max HP
  atk: number;            // Base attack
  def: number;            // Base defense
  critChance: number;     // Critical hit % (e.g. 12 = 12%)
  dodgeChance: number;    // Dodge % (max 30 via BattleScene.getUnitDodge)
  accuracy: number;       // Base accuracy (95-110 range)
  avatarUrl: string;      // Image URL
  aiType: "attacker" | "spellcaster" | "tank" | "boss";
  xpAwarded: number;      // XP reward
  goldAwarded: number;    // Gold reward
  dropMaterials: string[]; // Array of material IDs dropped on kill
}

[SPAWN FUNCTION] spawnEnemyGroup(chapter, area, isElite, isBoss, nodeLabel, way, path)
- Normal fight: 1-3 enemies (scales with area)
- Elite fight: 3 enemies with 1.4x stat multiplier
- Boss fight: single boss from CHAPTER_BOSSES
- hpMultiplier: scales from 0.6 to 1.0 across chapter areas
- progressionScaler: increases stats with area depth
